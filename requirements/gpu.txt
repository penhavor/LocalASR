# NVIDIA GPU acceleration (CUDA 12 + cuDNN 9) for the faster-whisper engine.

#   .venv\Scripts\pip install -r requirements/gpu.txt
#
# NOTE (Windows): besides these packages, the system cuBLAS/cuDNN 9 DLLs are
# required. The easiest source is the ready DLL archive from Purfview:

# unpack its contents into the environment's Lib\site-packages (or add to PATH).
#
# Linux: the pip packages below are enough (+ export LD_LIBRARY_PATH,
# LocalASR does it itself).

nvidia-cublas-cu12>=12.4
nvidia-cudnn-cu12>=9.1,<10
